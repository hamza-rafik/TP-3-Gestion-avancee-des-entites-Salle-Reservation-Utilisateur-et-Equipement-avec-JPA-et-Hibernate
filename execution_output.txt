[Hibernate] 
    create table equipements (
        id bigint generated by default as identity,
        code varchar(30) not null unique,
        etat varchar(30),
        nom varchar(100) not null,
        type varchar(50),
        primary key (id)
    )
[Hibernate] 
    create table reservations (
        date_debut timestamp(6) not null,
        date_fin timestamp(6) not null,
        id bigint generated by default as identity,
        motif varchar(150) not null,
        salle_id bigint not null,
        statut varchar(30) not null check (statut in ('CONFIRMEE','EN_ATTENTE','ANNULEE','TERMINEE')),
        utilisateur_id bigint not null,
        primary key (id)
    )
[Hibernate] 
    create table salle_equipement (
        equipement_id bigint not null,
        salle_id bigint not null,
        primary key (equipement_id, salle_id)
    )
[Hibernate] 
    create table salles (
        capacite integer not null,
        id bigint generated by default as identity,
        code varchar(30) not null unique,
        etage varchar(30),
        libelle varchar(100) not null,
        primary key (id)
    )
[Hibernate] 
    create table utilisateurs (
        id bigint generated by default as identity,
        email varchar(100) not null unique,
        nom varchar(50) not null,
        prenom varchar(50) not null,
        role varchar(50),
        telephone varchar(30),
        primary key (id)
    )
